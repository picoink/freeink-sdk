cmake_minimum_required(VERSION 3.20)
set(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(BOOK "${SDK_ROOT}/libs/book/FreeInkBook")
set(FONT "${SDK_ROOT}/libs/font/FreeInkFont")
set(UI "${SDK_ROOT}/libs/ui/FreeInkUI")
add_library(freeink_host_config INTERFACE)
target_compile_features(freeink_host_config INTERFACE cxx_std_20)
target_compile_definitions(freeink_host_config INTERFACE FREEINK_HOST=1 SIMULATOR=1
  FREEINK_DEVICE_METALIO_EINK4_PLUS=1 FREEINK_CAP_USB_MSC=1 EINK_DISPLAY_SINGLE_BUFFER_MODE=1)
target_include_directories(freeink_host_config INTERFACE
  "${CMAKE_CURRENT_LIST_DIR}/include" "${CMAKE_CURRENT_LIST_DIR}/third_party"
  "${BOOK}/include" "${FONT}/include" "${UI}/include"
  "${SDK_ROOT}/libs/assets/Icons/include" "${SDK_ROOT}/libs/assets/FreeInkImage/include" "${SDK_ROOT}/libs/assets/ProgressiveJpeg/include")
foreach(lib SDCardManager InputManager BoardConfig BatteryMonitor FrontlightManager HapticManager AudioManager Microphone Rtc Imu PowerManager UsbMassStorage)
  target_include_directories(freeink_host_config INTERFACE "${SDK_ROOT}/libs/hardware/${lib}/include")
endforeach()
foreach(lib expat miniz libunibreak pngle tjpgd stb)
  target_include_directories(freeink_host_config INTERFACE "${BOOK}/third_party/${lib}")
endforeach()
target_include_directories(freeink_host_config INTERFACE "${FONT}/third_party/stb" "${FONT}/third_party/freetype/include")
file(GLOB_RECURSE BOOK_SOURCES CONFIGURE_DEPENDS "${BOOK}/src/*.cpp" "${BOOK}/src/*.c")
file(GLOB FONT_SOURCES CONFIGURE_DEPENDS "${FONT}/src/*.cpp" "${FONT}/src/freetype/*.c")
find_package(JPEG REQUIRED)
add_library(freeink_reader_core STATIC src/GrayJpeg.c "${SDK_ROOT}/libs/assets/FreeInkImage/src/FreeInkImage.cpp" "${SDK_ROOT}/libs/assets/ProgressiveJpeg/src/ProgressiveJpeg.cpp" ${BOOK_SOURCES} ${FONT_SOURCES} "${FONT}/src/FontAlloc.c" "${UI}/src/FreeInkUI.cpp")
set_source_files_properties("${FONT}/src/FontAlloc.c" PROPERTIES COMPILE_DEFINITIONS ESP_PLATFORM=1)
target_link_libraries(freeink_reader_core PUBLIC freeink_host_config JPEG::JPEG)
if(PICO_STRESS)
  set(HOST_MEMORY_SOURCE src/MemoryStress.cpp)
  target_compile_definitions(freeink_host_config INTERFACE FREEINK_HOST_STRESS=1)
  target_compile_options(freeink_reader_core PRIVATE -fsanitize-coverage=trace-pc-guard)
else()
  set(HOST_MEMORY_SOURCE src/Memory.cpp)
endif()
add_library(freeink_host STATIC "${SDK_ROOT}/libs/hardware/BoardConfig/src/FlashAccess.cpp" "${SDK_ROOT}/libs/hardware/SDCardManager/src/VolumeStorage.cpp" src/Runtime.cpp src/Storage.cpp src/Preferences.cpp src/Devices.cpp src/Input.cpp ${HOST_MEMORY_SOURCE})
target_link_libraries(freeink_host PUBLIC freeink_host_config)
if(PICO_STRESS AND NOT APPLE)
  target_link_options(freeink_host INTERFACE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
endif()
add_executable(freeink_host_test tests/backend.cpp)
target_link_libraries(freeink_host_test PRIVATE freeink_host freeink_reader_core)
add_test(NAME host_backend COMMAND freeink_host_test)
add_executable(freeink_host_storage_test tests/storage.cpp)
target_link_libraries(freeink_host_storage_test PRIVATE freeink_host)
add_test(NAME host_storage COMMAND freeink_host_storage_test)
add_executable(freeink_host_volumes_test tests/volumes.cpp)
target_link_libraries(freeink_host_volumes_test PRIVATE freeink_host)
add_test(NAME host_volumes COMMAND freeink_host_volumes_test)
add_executable(freeink_host_coordinator_test tests/coordinator.cpp)
target_compile_options(freeink_host_coordinator_test PRIVATE -UNDEBUG)
target_link_libraries(freeink_host_coordinator_test PRIVATE freeink_host)
add_test(NAME host_resource_coordinator COMMAND freeink_host_coordinator_test)
add_executable(freeink_host_glyph_pages_test tests/glyph_pages.cpp)
target_compile_options(freeink_host_glyph_pages_test PRIVATE -UNDEBUG)
target_link_libraries(freeink_host_glyph_pages_test PRIVATE freeink_host)
add_test(NAME host_glyph_pages COMMAND freeink_host_glyph_pages_test)
if(PICO_STRESS)
  add_executable(freeink_host_external_task_test tests/external_task.cpp)
  target_compile_options(freeink_host_external_task_test PRIVATE -UNDEBUG)
  target_include_directories(freeink_host_external_task_test BEFORE PRIVATE "${CMAKE_CURRENT_LIST_DIR}/tests/task_stubs")
  target_link_libraries(freeink_host_external_task_test PRIVATE freeink_host)
  add_test(NAME host_external_task COMMAND freeink_host_external_task_test)
  add_executable(freeink_host_resources_test tests/resources.cpp)
  target_compile_options(freeink_host_resources_test PRIVATE -UNDEBUG)
  target_link_libraries(freeink_host_resources_test PRIVATE freeink_host)
  add_test(NAME host_resource_governance COMMAND freeink_host_resources_test)
  add_executable(freeink_host_memory_test tests/memory.cpp tests/allocation.c tests/cpu.cpp)
  target_compile_options(freeink_host_memory_test PRIVATE -UNDEBUG)
  set_source_files_properties(tests/cpu.cpp PROPERTIES COMPILE_OPTIONS -fsanitize-coverage=trace-pc-guard)
  target_link_libraries(freeink_host_memory_test PRIVATE freeink_host freeink_reader_core)
  add_test(NAME host_resources COMMAND freeink_host_memory_test)
endif()

add_executable(freeink_host_input_test tests/input.cpp)
target_compile_options(freeink_host_input_test PRIVATE -UNDEBUG)
target_link_libraries(freeink_host_input_test PRIVATE freeink_host)
add_test(NAME host_input COMMAND freeink_host_input_test)
